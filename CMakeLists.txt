cmake_minimum_required(VERSION 3.20)
project(launch-desk-obs VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
add_library(launch-core STATIC src/launcher.cpp src/dock.cpp)
target_link_libraries(launch-core PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets ole32 oleaut32 uuid shell32 advapi32)
target_compile_definitions(launch-core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(launch-core PUBLIC src)
add_library(launch-desk-obs MODULE src/plugin.cpp)
target_link_libraries(launch-desk-obs PRIVATE launch-core)
set_target_properties(launch-desk-obs PROPERTIES PREFIX "")
add_executable(launch-preview WIN32 tests/preview.cpp)
target_link_libraries(launch-preview PRIVATE launch-core)
add_executable(launch-tests tests/tests.cpp)
target_link_libraries(launch-tests PRIVATE launch-core)
add_executable(launch-fixture WIN32 tests/fixture.cpp)
add_library(frontend-test SHARED tests/frontend.cpp)
set_target_properties(frontend-test PROPERTIES OUTPUT_NAME "obs-frontend-api")
target_link_libraries(frontend-test PRIVATE Qt6::Widgets)
add_executable(launch-host-tests tests/host.cpp)
target_link_libraries(launch-host-tests PRIVATE launch-core)
foreach(target launch-core launch-desk-obs launch-preview launch-tests launch-fixture frontend-test launch-host-tests)
  target_compile_options(${target} PRIVATE /utf-8 /W4 /permissive- /Zc:__cplusplus)
endforeach()
enable_testing()
add_test(NAME launcher COMMAND launch-tests)
set_tests_properties(launcher PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal")
